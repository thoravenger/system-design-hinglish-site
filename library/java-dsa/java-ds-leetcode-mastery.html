// ── LeetCode: 217. Contains Duplicate ─────────────────── Easy ──────
public boolean containsDuplicate(int[] nums) {
    Set<Integer> seen = new HashSet<>();
    for (int n : nums) {
        if (!seen.add(n)) return true; // add returns false if already present
    }
    return false;
}

// ── LeetCode: 128. Longest Consecutive Sequence ────────── Hard ──────
public int longestConsecutive(int[] nums) {
    Set<Integer> set = new HashSet<>();
    for (int n : nums) set.add(n);

    int max = 0;
    for (int n : set) {
        if (!set.contains(n - 1)) {   // start of a sequence
            int curr = n, len = 1;
            while (set.contains(curr + 1)) { curr++; len++; }
            max = Math.max(max, len);
        }
    }
    return max;
}

// ── LeetCode: 3. Longest Substring Without Repeating Chars ── Medium ─
public int lengthOfLongestSubstring(String s) {
    Set<Character> window = new HashSet<>();
    int left = 0, max = 0;
    for (int right = 0; right < s.length(); right++) {
        while (window.contains(s.charAt(right))) {
            window.remove(s.charAt(left++));
        }
        window.add(s.charAt(right));
        max = Math.max(max, right - left + 1);
    }
    return max;
}

// ── LeetCode: 349. Intersection of Two Arrays ─────────── Easy ──────
public int[] intersection(int[] a, int[] b) {
    Set<Integer> setA = new HashSet<>();
    for (int n : a) setA.add(n);
    Set<Integer> res = new HashSet<>();
    for (int n : b) if (setA.contains(n)) res.add(n);
    return res.stream().mapToInt(Integer::intValue).toArray();
}

// ── Build & traverse ─────────────────────────────────────────────────
static ListNode build(int... vals) {
    ListNode dummy = new ListNode(0), cur = dummy;
    for (int v : vals) { cur.next = new ListNode(v); cur = cur.next; }
    return dummy.next;
}

static void print(ListNode head) {
    for (ListNode cur = head; cur != null; cur = cur.next)
        System.out.print(cur.val + (cur.next != null ? " → " : "\n"));
}

// ── LeetCode: 206. Reverse Linked List ────────────── Easy ───────────
// Iterative — O(n) time, O(1) space
public ListNode reverseList(ListNode head) {
    ListNode prev = null, curr = head;
    while (curr != null) {
        ListNode next = curr.next; // save next
        curr.next = prev;           // reverse pointer
        prev = curr;                // advance prev
        curr = next;                // advance curr
    }
    return prev;
}

// Recursive — O(n) time, O(n) stack space
public ListNode reverseListRec(ListNode head) {
    if (head == null || head.next == null) return head;
    ListNode newHead = reverseListRec(head.next);
    head.next.next = head;
    head.next = null;
    return newHead;
}

// ── LeetCode: 141. Linked List Cycle ──────────────── Easy ───────────
// Floyd's Two-Pointer (fast & slow) — O(n) time O(1) space
public boolean hasCycle(ListNode head) {
    ListNode slow = head, fast = head;
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
        if (slow == fast) return true; // they meet → cycle!
    }
    return false;
}

// ── LeetCode: 142. Linked List Cycle II — find cycle start ── Medium ─
public ListNode detectCycle(ListNode head) {
    ListNode slow = head, fast = head;
    while (fast != null && fast.next != null) {
        slow = slow.next; fast = fast.next.next;
        if (slow == fast) {
            ListNode ptr = head;
            while (ptr != slow) { ptr = ptr.next; slow = slow.next; }
            return ptr; // cycle start
        }
    }
    return null;
}

// ── LeetCode: 21. Merge Two Sorted Lists ─────────── Easy ───────────
public ListNode mergeTwoLists(ListNode l1, ListNode l2) {
    ListNode dummy = new ListNode(0), cur = dummy;
    while (l1 != null && l2 != null) {
        if (l1.val <= l2.val) { cur.next = l1; l1 = l1.next; }
        else                   { cur.next = l2; l2 = l2.next; }
        cur = cur.next;
    }
    cur.next = (l1 != null) ? l1 : l2;
    return dummy.next;
}

// ── LeetCode: 19. Remove Nth Node From End ──────── Medium ───────────
public ListNode removeNthFromEnd(ListNode head, int n) {
    ListNode dummy = new ListNode(0, head);
    ListNode fast = dummy, slow = dummy;
    for (int i = 0; i <= n; i++) fast = fast.next; // advance fast n+1 steps
    while (fast != null) { fast = fast.next; slow = slow.next; }
    slow.next = slow.next.next; // remove the nth node
    return dummy.next;
}

// ── LeetCode: 23. Merge K Sorted Lists ─────────── Hard ─────────────
public ListNode mergeKLists(ListNode[] lists) {
    PriorityQueue<ListNode> pq =
        new PriorityQueue<>(Comparator.comparingInt(n -> n.val));
    for (ListNode node : lists) if (node != null) pq.offer(node);

    ListNode dummy = new ListNode(0), cur = dummy;
    while (!pq.isEmpty()) {
        cur.next = pq.poll();
        cur = cur.next;
        if (cur.next != null) pq.offer(cur.next);
    }
    return dummy.next;
}

// ── LeetCode: 146. LRU Cache ─────────────────────── Medium ─────────
// Uses: HashMap + Doubly Linked List (O(1) get + put)
class LRUCache {

    private static class Node {
        int key, val;
        Node prev, next;
        Node(int k, int v) { key = k; val = v; }
    }

    private final int               cap;
    private final Map<Integer, Node>  map  = new HashMap<>();
    private final Node               head = new Node(0, 0); // dummy head
    private final Node               tail = new Node(0, 0); // dummy tail

    LRUCache(int capacity) {
        cap = capacity;
        head.next = tail; tail.prev = head;  // sentinel doubly linked list
    }

    public int get(int key) {
        if (!map.containsKey(key)) return -1;
        Node n = map.get(key);
        remove(n); addToFront(n); // move to most recently used
        return n.val;
    }

    public void put(int key, int value) {
        if (map.containsKey(key)) remove(map.get(key));
        Node n = new Node(key, value);
        map.put(key, n); addToFront(n);
        if (map.size() > cap) {
            Node lru = tail.prev;
            remove(lru);
            map.remove(lru.key);
        }
    }

    private void remove(Node n) {
        n.prev.next = n.next; n.next.prev = n.prev;
    }
    private void addToFront(Node n) {
        n.next = head.next; n.prev = head;
        head.next.prev = n; head.next = n;
    }
}

// Prefer offer/poll/peek — they return null/false instead of throwing

// Enqueue
q.offer("first");   // returns false if full (non-throwing)
q.add("second");   // throws IllegalStateException if full

// Dequeue

String head2 = q.remove(); // throws NoSuchElementException if empty

// ── LeetCode: 102. Binary Tree Level Order Traversal ── Medium ───────
// BFS with Queue — the most common Queue pattern
public List<List<Integer>> levelOrder(TreeNode root) {
    List<List<Integer>> result = new ArrayList<>();
    if (root == null) return result;

    Queue<TreeNode> q = new ArrayDeque<>();
    q.offer(root);

    while (!q.isEmpty()) {
        int size = q.size();              // snapshot this level's count
        List<Integer> level = new ArrayList<>();
        for (int i = 0; i < size; i++) {
            TreeNode node = q.poll();
            level.add(node.val);
            if (node.left  != null) q.offer(node.left);
            if (node.right != null) q.offer(node.right);
        }
        result.add(level);
    }
    return result;
}

// ── LeetCode: 239. Sliding Window Maximum ────────── Hard ────────────
// Monotonic Deque — O(n) time
public int[] maxSlidingWindow(int[] nums, int k) {
    Deque<Integer> deque = new ArrayDeque<>(); // stores INDICES
    int[] res = new int[nums.length - k + 1];

    for (int i = 0; i < nums.length; i++) {
        // Remove elements outside window
        while (!deque.isEmpty() && deque.peekFirst() < i - k + 1)
            deque.pollFirst();
        // Maintain decreasing deque
        while (!deque.isEmpty() && nums[deque.peekLast()] < nums[i])
            deque.pollLast();

        deque.offerLast(i);
        if (i >= k - 1) res[i - k + 1] = nums[deque.peekFirst()];
    }
    return res;
}

// ── LeetCode: 215. Kth Largest Element ─────────── Medium ───────────
// Min-heap of size k — O(n log k)
public int findKthLargest(int[] nums, int k) {
    PriorityQueue<Integer> minH = new PriorityQueue<>();
    for (int n : nums) {
        minH.offer(n);
        if (minH.size() > k) minH.poll(); // evict smallest
    }
    return minH.peek(); // top of min-heap = kth largest
}

// ── LeetCode: 295. Find Median from Data Stream ─── Hard ────────────
// Two heaps: maxHeap (lower half) + minHeap (upper half)
class MedianFinder {
    PriorityQueue<Integer> lo = new PriorityQueue<>(Collections.reverseOrder()); // max-heap
    PriorityQueue<Integer> hi = new PriorityQueue<>();                             // min-heap

    public void addNum(int num) {
        lo.offer(num);
        hi.offer(lo.poll());           // balance: push lo's max to hi
        if (lo.size() < hi.size()) lo.offer(hi.poll()); // keep lo ≥ hi size
    }

    public double findMedian() {
        return lo.size() > hi.size()
            ? lo.peek()
            : (lo.peek() + hi.peek()) / 2.0;
    }
}

// ── LeetCode: 200. Number of Islands ─────────────── Medium ─────────
// BFS with Queue — flood fill pattern
public int numIslands(char[][] grid) {
    int islands = 0;
    int[] dx = {0,0,1,-1}, dy = {1,-1,0,0};

    for (int r = 0; r < grid.length; r++) {
        for (int c = 0; c < grid[0].length; c++) {
            if (grid[r][c] == '1') {
                islands++;
                Queue<int[]> q = new ArrayDeque<>();
                q.offer(new int[]{r, c});
                grid[r][c] = '0'; // mark visited
                while (!q.isEmpty()) {
                    int[] cur = q.poll();
                    for (int d = 0; d < 4; d++) {
                        int nr = cur[0] + dx[d], nc = cur[1] + dy[d];
                        if (nr >= 0 && nr < grid.length && nc >= 0
                                && nc < grid[0].length && grid[nr][nc] == '1') {
                            grid[nr][nc] = '0';
                            q.offer(new int[]{nr, nc});
                        }
                    }
                }
            }
        }
    }
    return islands;
}

// ── 1. Classic Producer-Consumer with BlockingQueue ──────────────────

public class ProducerConsumer {

    private static final int    CAPACITY = 10;
    private static final Object POISON   = new Object(); // sentinel

    private final BlockingQueue<Object> queue =
        new LinkedBlockingQueue<>(CAPACITY);

    class Producer implements Runnable {
        private final int id;
        Producer(int id) { this.id = id; }

        @Override
        public void run() {
            try {
                for (int i = 0; i < 5; i++) {
                    Object item = "P" + id + "-item-" + i;
                    queue.put(item);   // BLOCKS if queue is full
                    log.info("Produced: {}", item);
                }
                queue.put(POISON);    // signal done
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }
    }

    class Consumer implements Runnable {
        @Override
        public void run() {
            try {
                while (true) {
                    Object item = queue.take();  // BLOCKS if queue is empty
                    if (item == POISON) break;   // received shutdown signal
                    process(item);
                    log.info("Consumed: {}", item);
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }
    }

    public void run(int producerCount, int consumerCount) throws InterruptedException {
        List<Thread> all = new ArrayList<>();
        IntStream.range(0, producerCount)
            .forEach(i -> all.add(Thread.ofVirtual().start(new Producer(i))));
        IntStream.range(0, consumerCount)
            .forEach(i -> all.add(Thread.ofVirtual().start(new Consumer())));
        for (Thread t : all) t.join();
    }
}

// ── 4. Rate Limiter using DelayQueue ─────────────────────────────────

class RateLimiter {

    static class Token implements Delayed {
        private final long availableAt;
        Token(long delayMs) { availableAt = System.currentTimeMillis() + delayMs; }

        @Override public long getDelay(TimeUnit unit) {
            return unit.convert(availableAt - System.currentTimeMillis(), TimeUnit.MILLISECONDS);
        }
        @Override public int compareTo(Delayed o) {
            return Long.compare(this.availableAt, ((Token)o).availableAt);
        }
    }

    private final DelayQueue<Token> tokens = new DelayQueue<>();
    private final int ratePerSecond;

    RateLimiter(int ratePerSecond) {
        this.ratePerSecond = ratePerSecond;
        // pre-fill tokens
        IntStream.range(0, ratePerSecond).forEach(i -> tokens.offer(new Token(0)));
    }

    public void acquire() throws InterruptedException {
        tokens.take(); // blocks until a token's delay expires
        // schedule next token 1 second from now
        tokens.offer(new Token(1000));
    }
}

// ── 5. Bounded Buffer (wait/notify pattern — classic interview Q) ─────

public class BoundedBuffer<T> {

    private final Queue<T> buffer;
    private final int      capacity;

    BoundedBuffer(int capacity) {
        this.capacity = capacity;
        this.buffer   = new ArrayDeque<>(capacity);
    }

    public synchronized void put(T item) throws InterruptedException {
        while (buffer.size() == capacity) {
            wait(); // release lock, wait until space available
        }
        buffer.offer(item);
        notifyAll(); // wake waiting consumers
    }

    public synchronized T take() throws InterruptedException {
        while (buffer.isEmpty()) {
            wait(); // release lock, wait until item available
        }
        T item = buffer.poll();
        notifyAll(); // wake waiting producers
        return item;
    }
}

// ── Build a ThreadPool from scratch using BlockingQueue ──────────────

public class SimpleThreadPool {

    private final BlockingQueue<Runnable> workQueue;
    private final List<Worker>           workers;
    private volatile boolean              shutdown = false;

    public SimpleThreadPool(int threads, int queueCapacity) {
        workQueue = new LinkedBlockingQueue<>(queueCapacity);
        workers   = IntStream.range(0, threads)
            .mapToObj(i -> new Worker("pool-worker-" + i))
            .collect(Collectors.toList());
        workers.forEach(Thread::start);
    }

    public void submit(Runnable task) {
        if (shutdown) throw new RejectedExecutionException("Pool is shut down");
        if (!workQueue.offer(task)) {
            throw new RejectedExecutionException("Queue is full");
        }
    }

    public void shutdown() {
        shutdown = true;
        workers.forEach(Thread::interrupt);
    }

    private class Worker extends Thread {
        Worker(String name) {
            super(name);
            setDaemon(true); // dies when main thread dies
        }

        @Override
        public void run() {
            while (!isInterrupted()) {
                try {
                    Runnable task = workQueue.poll(1, TimeUnit.SECONDS);
                    if (task != null) task.run();
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt(); // restore flag
                } catch (Exception e) {
                    log.error("Task failed in {}: {}", getName(), e.getMessage());
                    // Don't let one bad task kill the worker thread
                }
            }
        }
    }
}

SimpleThreadPool pool = new SimpleThreadPool(4, 100);
IntStream.range(0, 20).forEach(i ->

// ── Work-Stealing ThreadPool (ForkJoinPool) — Java 21 ────────────────
// Each worker has its own deque. Idle workers STEAL from busy ones.

ForkJoinPool fjPool = new ForkJoinPool(4); // 4 worker threads

// Submit recursive task (divide-and-conquer)
class SumTask extends RecursiveTask<Long> {
    private final int[] arr;
    private final int   lo, hi;
    static final int    THRESHOLD = 1000;

    SumTask(int[] arr, int lo, int hi) { this.arr=arr; this.lo=lo; this.hi=hi; }

    @Override
    protected Long compute() {
        if (hi - lo <= THRESHOLD) {
            long sum = 0;
            for (int i = lo; i < hi; i++) sum += arr[i];
            return sum;
        }
        int mid = (lo + hi) / 2;
        SumTask left  = new SumTask(arr, lo, mid);
        SumTask right = new SumTask(arr, mid, hi);
        left.fork();                    // push left to own deque (async)
        return right.compute() + left.join(); // right inline, then get left
    }
}

int[] data = IntStream.range(0, 1_000_000).toArray();
long total = fjPool.invoke(new SumTask(data, 0, data.length));

// ── LeetCode: 1. Two Sum ──────────────────────────── Easy ───────────
public int[] twoSum(int[] nums, int target) {
    Map<Integer, Integer> map = new HashMap<>(); // val → index
    for (int i = 0; i < nums.length; i++) {
        int complement = target - nums[i];
        if (map.containsKey(complement))
            return new int[]{map.get(complement), i};
        map.put(nums[i], i);
    }
    return new int[]{};
}

// ── LeetCode: 347. Top K Frequent Elements ───────── Medium ─────────
public int[] topKFrequent(int[] nums, int k) {
    Map<Integer, Integer> freq = new HashMap<>();
    for (int n : nums) freq.merge(n, 1, Integer::sum);

    PriorityQueue<Integer> minH =
        new PriorityQueue<>(Comparator.comparingInt(freq::get));
    for (int key : freq.keySet()) {
        minH.offer(key);
        if (minH.size() > k) minH.poll();
    }
    return minH.stream().mapToInt(Integer::intValue).toArray();
}

// ── LeetCode: 560. Subarray Sum Equals K ─────────── Medium ─────────
// Prefix sum + HashMap — O(n)
public int subarraySum(int[] nums, int k) {
    Map<Integer, Integer> prefixCount = new HashMap<>();
    prefixCount.put(0, 1); // empty prefix
    int sum = 0, result = 0;
    for (int n : nums) {
        sum += n;
        result += prefixCount.getOrDefault(sum - k, 0);
        prefixCount.merge(sum, 1, Integer::sum);
    }
    return result;
}

// ── LeetCode: 76. Minimum Window Substring ────────── Hard ───────────
// Sliding window with HashMap — O(n)
public String minWindow(String s, String t) {
    Map<Character, Integer> need = new HashMap<>();
    for (char c : t.toCharArray()) need.merge(c, 1, Integer::sum);

    Map<Character, Integer> window = new HashMap<>();
    int have = 0, need_cnt = need.size();
    int[] ans = {-1, 0, 0}; // length, left, right
    int left = 0;

    for (int right = 0; right < s.length(); right++) {
        char c = s.charAt(right);
        window.merge(c, 1, Integer::sum);
        if (need.containsKey(c) && window.get(c).equals(need.get(c))) have++;

        while (have == need_cnt) {
            if (ans[0] == -1 || right - left + 1 < ans[0])
                ans = new int[]{right - left + 1, left, right};
            char lc = s.charAt(left++);
            window.merge(lc, -1, Integer::sum);
            if (need.containsKey(lc) && window.get(lc) < need.get(lc)) have--;
        }
    }
    return ans[0] == -1 ? "" : s.substring(ans[1], ans[2] + 1);
}

// ── Pattern: Monotonic Stack — Next Greater Element ──────────────────
// LeetCode 496. Next Greater Element I — Easy
public int[] nextGreaterElement(int[] nums1, int[] nums2) {
    Map<Integer, Integer> nge = new HashMap<>();
    Deque<Integer> stack = new ArrayDeque<>(); // stores nums2 values

    for (int n : nums2) {
        while (!stack.isEmpty() && stack.peek() < n)
            nge.put(stack.pop(), n);          // n is the next greater
        stack.push(n);
    }
    return Arrays.stream(nums1)
        .map(n -> nge.getOrDefault(n, -1))
        .toArray();
}

// ── Pattern: Trie — Word Search ──────────────────────────────────────
// LeetCode 208. Implement Trie — Medium
class Trie {
    private Map<Character, Trie> children = new HashMap<>();
    private boolean isEnd = false;

    public void insert(String word) {
        Trie node = this;
        for (char c : word.toCharArray()) {
            node.children.putIfAbsent(c, new Trie());
            node = node.children.get(c);
        }
        node.isEnd = true;
    }

    public boolean search(String word) {
        Trie node = find(word);
        return node != null && node.isEnd;
    }

    public boolean startsWith(String prefix) {
        return find(prefix) != null;
    }

    private Trie find(String s) {
        Trie node = this;
        for (char c : s.toCharArray()) {
            if (!node.children.containsKey(c)) return null;
            node = node.children.get(c);
        }
        return node;
    }
}

// ── Pattern: Union-Find (Disjoint Set) ───────────────────────────────
// LeetCode 200. Number of Islands — also solvable with Union-Find
class UnionFind {
    private final int[] parent, rank;

    UnionFind(int n) {
        parent = new int[n];
        rank   = new int[n];
        Arrays.setAll(parent, i -> i); // each node is its own parent
    }

    public int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]); // path compression
        return parent[x];
    }

    public boolean union(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return false; // already connected
        if (rank[px] < rank[py]) { int t = px; px = py; py = t; }
        parent[py] = px;
        if (rank[px] == rank[py]) rank[px]++;
        return true;
    }
}

// ── LeetCode: 621. Task Scheduler ────────────── Medium ──────────────
// PriorityQueue (max-heap) + Queue (cooldown) — greedy simulation
public int leastInterval(char[] tasks, int n) {
    int[] freq = new int[26];
    for (char c : tasks) freq[c - 'A']++;

    // Max-heap: always pick most frequent available task
    PriorityQueue<Integer> maxH = new PriorityQueue<>(Collections.reverseOrder());
    for (int f : freq) if (f > 0) maxH.offer(f);

    // Queue: [remaining_count, time_when_available]
    Queue<int[]> cooldown = new ArrayDeque<>();
    int time = 0;

    while (!maxH.isEmpty() || !cooldown.isEmpty()) {
        time++;
        if (!maxH.isEmpty()) {
            int remaining = maxH.poll() - 1;
            if (remaining > 0) cooldown.offer(new int[]{remaining, time + n});
        }
        if (!cooldown.isEmpty() && cooldown.peek()[1] == time)
            maxH.offer(cooldown.poll()[0]);
    }
    return time;
}

// ── LeetCode: 380. Insert Delete GetRandom O(1) ── Medium ───────────
class RandomizedSet {
    private final Map<Integer, Integer> map  = new HashMap<>(); // val → index
    private final List<Integer>        list = new ArrayList<>();
    private final Random               rand = new Random();

    public boolean insert(int val) {
        if (map.containsKey(val)) return false;
        list.add(val);
        map.put(val, list.size() - 1);
        return true;
    }

    public boolean remove(int val) {
        if (!map.containsKey(val)) return false;
        int idx  = map.get(val);
        int last = list.get(list.size() - 1);
        list.set(idx, last);                  // swap with last
        map.put(last, idx);                    // update last's index
        list.remove(list.size() - 1);         // remove last O(1)
        map.remove(val);
        return true;
    }

    public int getRandom() {
        return list.get(rand.nextInt(list.size())); // O(1) random access
    }
}

// ── LeetCode: 895. Maximum Frequency Stack ────── Hard ───────────────
// Map of stacks keyed by frequency
class FreqStack {
    private final Map<Integer, Integer>        freq    = new HashMap<>();
    private final Map<Integer, Deque<Integer>> group   = new HashMap<>();
    private int                                  maxFreq = 0;

    public void push(int val) {
        int f = freq.merge(val, 1, Integer::sum);
        maxFreq = Math.max(maxFreq, f);
        group.computeIfAbsent(f, k -> new ArrayDeque<>()).push(val);
    }

    public int pop() {
        Deque<Integer> stack = group.get(maxFreq);
        int val = stack.pop();
        if (stack.isEmpty()) { group.remove(maxFreq); maxFreq--; }
        freq.merge(val, -1, Integer::sum);
        return val;
    }
}
