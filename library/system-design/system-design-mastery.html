// ── Algorithm 1: Fixed Window Counter ────────────────────────────────
// Redis key: "rl:{userId}:{windowStart}"
// Atomic INCR + EXPIRE

public boolean isAllowed_FixedWindow(String userId, int limit, int windowSec) {
    String window = String.valueOf(System.currentTimeMillis() / (windowSec * 1000L));
    String key    = "rl:fixed:" + userId + ":" + window;

    Long count = redis.execute(new DefaultRedisScript<>("""
        local count = redis.call('INCR', KEYS[1])
        if count == 1 then
            redis.call('EXPIRE', KEYS[1], ARGV[1])
        end
        return count
        """, Long.class),
        List.of(key), String.valueOf(windowSec)
    );
    return count != null && count <= limit;
}

Algorithm 2 — Sliding Window Log

// ── Algorithm 2: Sliding Window Log ──────────────────────────────────
// Store timestamp of each request in a Redis sorted set
// Score = timestamp. Remove entries older than windowSec.
// Count remaining entries. Accept if under limit.
// Memory: O(limit) per user — accurate but uses more memory

public boolean isAllowed_SlidingLog(String userId, int limit, int windowMs) {
    String key  = "rl:log:" + userId;
    long   now  = System.currentTimeMillis();
    long   from = now - windowMs;

    Long count = redis.execute(new DefaultRedisScript<>("""
        local key    = KEYS[1]
        local now    = tonumber(ARGV[1])
        local from   = tonumber(ARGV[2])
        local limit  = tonumber(ARGV[3])
        local window = tonumber(ARGV[4])
        redis.call('ZREMRANGEBYSCORE', key, '-inf', from)   -- remove old
        local count = redis.call('ZCARD', key)
        if count < limit then
            redis.call('ZADD', key, now, now)               -- add current
            redis.call('EXPIRE', key, window / 1000 + 1)
            return 1
        end
        return 0
        """, Long.class),
        List.of(key),
        String.valueOf(now),
        String.valueOf(from),
        String.valueOf(limit),
        String.valueOf(windowMs)
    );
    return Long.valueOf(1).equals(count);
}

Algorithm 3 — Sliding Window Counter (Hybrid — Best for Production)

// ── Algorithm 3: Sliding Window Counter ───────────────────────────────
// Approximation: blend current window + previous window
// count = prev_count * overlap_ratio + curr_count
// Example: 60s window, at 15s into new window:
//   overlap = (60-15)/60 = 0.75
//   approx_count = prev(40) * 0.75 + curr(5) = 35
// Memory: O(1) — only 2 counters needed

public boolean isAllowed_SlidingCounter(String userId, int limit, int windowSec) {
    long   now       = System.currentTimeMillis() / 1000L;
    long   currWindow = now / windowSec;
    long   prevWindow = currWindow - 1;
    double elapsed    = (now % windowSec) / (double) windowSec;

    String currKey = "rl:sc:" + userId + ":" + currWindow;
    String prevKey = "rl:sc:" + userId + ":" + prevWindow;

    Long   currCount = redis.opsForValue().increment(currKey);
    Object prevObj   = redis.opsForValue().get(prevKey);
    long   prevCount = prevObj != null ? Long.parseLong(prevObj.toString()) : 0L;

    redis.expire(currKey, Duration.ofSeconds(windowSec * 2L));

    double approx = prevCount * (1.0 - elapsed) + currCount;
    return approx <= limit;
}

// ── Algorithm 4: Token Bucket (Redis Lua) ────────────────────────────
// Each request takes 1 token. Tokens refill at a steady rate.
// Allows burst up to bucket capacity.

private static final String TOKEN_BUCKET_SCRIPT = """
    local key          = KEYS[1]
    local capacity     = tonumber(ARGV[1])
    local refill_rate  = tonumber(ARGV[2])  -- tokens per second
    local now          = tonumber(ARGV[3])  -- epoch milliseconds
    local requested    = tonumber(ARGV[4])  -- tokens needed (usually 1)

    local data = redis.call('HMGET', key, 'tokens', 'last_refill')
    local tokens      = tonumber(data[1]) or capacity
    local last_refill = tonumber(data[2]) or now

    -- Compute tokens to add based on elapsed time
    local elapsed = (now - last_refill) / 1000.0
    tokens = math.min(capacity, tokens + elapsed * refill_rate)

    if tokens >= requested then
        tokens = tokens - requested
        redis.call('HMSET', key, 'tokens', tokens, 'last_refill', now)
        redis.call('EXPIRE', key, 3600)
        return 1   -- allowed
    else
        redis.call('HMSET', key, 'tokens', tokens, 'last_refill', now)
        redis.call('EXPIRE', key, 3600)
        return 0   -- denied
    end
    """;

public boolean isAllowed_TokenBucket(String userId, int capacity, int refillPerSec) {
    DefaultRedisScript<Long> script = new DefaultRedisScript<>(TOKEN_BUCKET_SCRIPT, Long.class);
    Long result = redis.execute(script,
        List.of("rl:tb:" + userId),
        String.valueOf(capacity),
        String.valueOf(refillPerSec),
        String.valueOf(System.currentTimeMillis()),
        "1"
    );
    return Long.valueOf(1).equals(result);
}

Algorithm 5 — Leaky Bucket

// ── Algorithm 5: Leaky Bucket ─────────────────────────────────────────
// Requests added to queue (bucket). Processed at constant rate (leak).
// Smooths out bursts — output is always at fixed rate.
// If bucket full → request DROPPED.
// Use case: network bandwidth shaping, metered billing

public class LeakyBucket {
    private final BlockingQueue<Runnable> bucket;
    private final ScheduledExecutorService scheduler;

    public LeakyBucket(int capacity, int leakRatePerSec) {
        bucket    = new ArrayBlockingQueue<>(capacity);
        scheduler = Executors.newScheduledThreadPool(1);
        scheduler.scheduleAtFixedRate(this::leak,
            0, 1000L / leakRatePerSec, TimeUnit.MILLISECONDS);
    }

    public boolean submit(Runnable request) {
        return bucket.offer(request); // false if bucket full → drop request
    }

    private void leak() {
        Runnable req = bucket.poll();
        if (req != null) req.run(); // process at constant rate
    }
}

// ── Spring Boot Rate Limit Filter ────────────────────────────────────
@Component
@Order(Ordered.HIGHEST_PRECEDENCE)
public class RateLimitFilter implements Filter {

    private final RateLimiterService rateLimiter;

    @Override
    public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain)
            throws IOException, ServletException {

        HttpServletRequest  httpReq = (HttpServletRequest)  req;
        HttpServletResponse httpRes = (HttpServletResponse) res;

        String userId = extractUserId(httpReq);  // from JWT / API key
        if (!rateLimiter.isAllowed(userId)) {
            httpRes.setStatus(429);             // 429 Too Many Requests
            httpRes.setHeader("Retry-After", "60");
            httpRes.setHeader("X-RateLimit-Limit", "100");
            httpRes.setHeader("X-RateLimit-Remaining", "0");
            httpRes.getWriter().write("""
                {"error":"rate_limit_exceeded","message":"Too many requests"}
                """);
            return;
        }
        chain.doFilter(req, res);
    }
}

A Bloom filter answers: "Is this element definitely NOT in the set, or POSSIBLY in the set?" — space efficient, O(1), but has false positives (never false negatives).

// ── Bloom Filter — Java implementation ───────────────────────────────

public class BloomFilter {
    private final BitSet bitSet;
    private final int    size;
    private final int    hashFunctions;

    // Optimal params: m = -n*ln(p)/(ln2)^2, k = (m/n)*ln2
    // n = expected elements, p = false positive rate
    // For n=1M, p=1%: m≈9.6M bits (1.2MB), k≈7 hash functions
    public BloomFilter(int size, int hashFunctions) {
        this.size          = size;
        this.hashFunctions = hashFunctions;
        this.bitSet        = new BitSet(size);
    }

    public void add(String item) {
        for (int i = 0; i < hashFunctions; i++) {
            int pos = hash(item, i) % size;
            bitSet.set(Math.abs(pos));
        }
    }

    public boolean mightContain(String item) {
        for (int i = 0; i < hashFunctions; i++) {
            int pos = hash(item, i) % size;
            if (!bitSet.get(Math.abs(pos))) return false; // definitely NOT in set
        }
        return true; // possibly in set (may be false positive)
    }

    private int hash(String item, int seed) {
        return Objects.hash(item, seed); // in prod: use MurmurHash3 with different seeds
    }
}

All operations in a transaction succeed or all fail. No partial state.

Consistency

Isolation

-- ══ Index Types ══════════════════════════════════════════════════════

-- B-Tree (default) — works for: =, <, >, BETWEEN, LIKE 'prefix%'
CREATE INDEX idx_orders_user   ON orders(user_id);
CREATE INDEX idx_orders_status ON orders(status, created_at DESC); -- composite
CREATE UNIQUE INDEX idx_users_email ON users(email);

-- Partial Index — only index rows matching condition (smaller, faster)
CREATE INDEX idx_active_orders ON orders(user_id)
    WHERE status IN ('PENDING', 'PROCESSING');  -- only index active orders

-- Covering Index — index contains all columns needed (no heap fetch)
CREATE INDEX idx_orders_cover ON orders(user_id, status) INCLUDE (total, created_at);
-- Query: SELECT total FROM orders WHERE user_id=1 → index-only scan (no table hit)

-- Hash Index — O(1) exact match only (no range, no sorting)
CREATE INDEX idx_hash_email ON users USING hash(email);

-- GIN — full-text search, arrays, JSONB
CREATE INDEX idx_tags  ON products USING gin(tags);    -- tags text[]
CREATE INDEX idx_props ON products USING gin(props jsonb_path_ops); -- JSONB

-- GiST — geometric, full-text, range types
CREATE INDEX idx_location ON stores USING gist(location);  -- PostGIS geometry

-- BRIN (Block Range INdex) — huge sequential tables (logs, time-series)
CREATE INDEX idx_created_brin ON events USING brin(created_at);
-- 1000x smaller than B-tree, works because rows are physically ordered by time

-- Expression Index — index on function result
CREATE INDEX idx_email_lower ON users(LOWER(email));
-- Query: WHERE LOWER(email) = 'alice@example.com' → uses index

Database Sharding — Strategies & Trade-offs

Sharding (horizontal partitioning) splits data across multiple databases so no single database has the full dataset. Each shard is an independent DB holding a subset of rows.

Sharding Architecture
App Server
→
Shard Router (Proxy)
→
Shard 1 (user 0-25%)
→
Shard 2 (user 25-50%)
→
Shard 3 (user 50-75%)
→
Shard 4 (user 75-100%)
// ── Sharding Strategy 1: Range Sharding ──────────────────────────────
// userId 1-1M       → Shard 1
// userId 1M-2M      → Shard 2
// userId 2M-3M      → Shard 3
//
// ✅ Easy range queries (get all users in ID range)
// ❌ Hot spots — new users always go to last shard
// ❌ Uneven load if distribution not uniform

int getShardId_Range(long userId) {
    if (userId < 1_000_000L)  return 1;
    if (userId < 2_000_000L)  return 2;
    if (userId < 3_000_000L)  return 3;
    return 4;
}

// ── Sharding Strategy 2: Hash Sharding ───────────────────────────────
// shardId = hash(userId) % numShards
//
// ✅ Even distribution
// ❌ Cross-shard queries expensive
// ❌ Adding shards requires resharding ALL data (use consistent hashing!)

int getShardId_Hash(long userId, int numShards) {
    return (int) (Math.abs(Objects.hash(userId)) % numShards);
}

// ── Sharding Strategy 3: Directory Sharding ──────────────────────────
// A lookup service maps entity → shard
// userId 1 → Shard 3 (stored in lookup DB)
//
// ✅ Most flexible — move data between shards without changing logic
// ❌ Lookup service is a single point of failure + extra latency

@Service
public class ShardRouter {
    private final Map<Long, DataSource> shardMap = new ConcurrentHashMap<>();

    public DataSource getShardForUser(long userId) {
        int shardId = (int) (Math.abs(userId) % 4);
        return shardMap.get((long) shardId);
    }
}

// ── Shard Key choice — critical decisions ─────────────────────────────
// ✅ Good shard keys: userId, tenantId — high cardinality, uniform
// ❌ Bad shard keys: country (hot: US), status (hot: ACTIVE), created_at (hot: latest)
//
// Compound shard key: (tenantId, userId) → tenant isolation within shard
//
// Cross-shard problems:
// - JOIN across shards → collect from all shards, merge in app layer
// - ORDER BY + LIMIT → fetch from all shards, merge-sort, take top K
// - COUNT(*) → sum counts from all shards
// - Transactions → distributed transaction (2PC) or saga pattern

Java 17 — Sharding
✅ When to Shard
  • Single DB > 1-2 TB
  • Write throughput maxed out
  • Can't scale vertically any more
  • Multi-tenancy (tenant-per-shard)
  • Regulatory data residency requirements
⚠️ Before Sharding
  • Read replicas for read scale
  • Vertical scaling (better hardware)
  • Caching (Redis) — reduce DB load
  • Database partitioning (within one DB)
  • Query & index optimization
❌ Sharding Problems
  • Cross-shard JOINs are hard
  • Distributed transactions (saga needed)
  • Resharding is painful
  • Shard hotspots (celebrity problem)
  • Operational complexity 10x higher

// ── Spring Boot: Route writes to primary, reads to replica ───────────

public enum DbType { PRIMARY, REPLICA }

public class ReplicationRoutingDataSource extends AbstractRoutingDataSource {

    private static final ThreadLocal<DbType> CONTEXT =
        ThreadLocal.withInitial(() -> DbType.PRIMARY);

    public static void usePrimary()  { CONTEXT.set(DbType.PRIMARY); }
    public static void useReplica()  { CONTEXT.set(DbType.REPLICA); }
    public static void clear()       { CONTEXT.remove(); }

    @Override
    protected Object determineCurrentLookupKey() {
        return CONTEXT.get();
    }
}

// AOP: @Transactional(readOnly=true) → replica, else → primary
@Aspect
@Component
public class DataSourceAspect {

    @Around("@annotation(transactional)")
    public Object routeDataSource(ProceedingJoinPoint pjp, Transactional transactional)
            throws Throwable {
        try {
            if (transactional.readOnly()) {
                ReplicationRoutingDataSource.useReplica();
            } else {
                ReplicationRoutingDataSource.usePrimary();
            }
            return pjp.proceed();
        } finally {
            ReplicationRoutingDataSource.clear();
        }
    }
}

Write Primary → Read Replica: How to Keep in Sync

⚠️
The problem: User writes to primary. User immediately reads. Request goes to replica. Replica has 50ms lag. User sees OLD data. This is called Read-Your-Writes Inconsistency — and it's the most common complaint in systems with replication.
// ── Pattern 1: Read-Your-Writes — route post-write reads to primary ───
// After a write, stick to primary for reads for a short window

@Service
public class ReadYourWritesService {

    private final Cache<String, Long> recentWrites = Caffeine.newBuilder()
        .expireAfterWrite(Duration.ofSeconds(5))  // assume replica catches up in 5s
        .build();

    public void updateProfile(String userId, UserProfile profile) {
        // Write to primary
        ReplicationRoutingDataSource.usePrimary();
        userRepo.save(profile);

        // Remember this user wrote recently
        recentWrites.put(userId, System.currentTimeMillis());
    }

    public UserProfile getProfile(String userId) {
        // If user wrote recently → read from primary (fresh data)
        if (recentWrites.getIfPresent(userId) != null) {
            ReplicationRoutingDataSource.usePrimary();
        } else {
            ReplicationRoutingDataSource.useReplica();  // safe to use replica
        }
        return userRepo.findById(userId).orElseThrow();
    }
}

// ── Pattern 2: Monotonic Reads — same user always same replica ────────
// Route each user to a specific replica — they see monotonically increasing state
// userId → replicaId (via hash or sticky session)

int getReplicaForUser(long userId, int replicaCount) {
    return (int) (userId % replicaCount);  // always same replica for this user
}

// ── Pattern 3: Write to cache on write, read from cache ───────────────
// Write-Through Cache: write to DB + cache atomically

@Service
public class WriteThroughService {

    public void updateUser(User user) {
        userRepo.save(user);                                     // write to primary DB
        redis.opsForValue().set("user:" + user.getId(), user,
            Duration.ofMinutes(10));                             // update cache
    }

    public User getUser(long userId) {
        User cached = (User) redis.opsForValue().get("user:" + userId);
        if (cached != null) return cached;     // fresh from cache (beats replication lag)
        return userRepo.findById(userId).orElseThrow(); // read from replica/primary
    }
}

// ── Pattern 4: Synchronous Replication (for critical reads) ──────────
// PostgreSQL synchronous_commit = on/remote_apply
// Primary WAITS for replica to apply WAL before confirming write
// Guarantees: zero replication lag
// Cost: write latency increases by network RTT to replica

-- PostgreSQL: synchronous_standby_names = 'replica1'
-- Per-transaction override:
SET synchronous_commit = remote_apply;
INSERT INTO critical_data VALUES (...);  -- replica guaranteed to have this
RESET synchronous_commit;

// ── Pattern 5: Version / Timestamp comparison ─────────────────────────
// Add version column. Client stores version after write.
// On read, if version matches or is newer → replica is fine
// If version on replica is older → escalate to primary

@Transactional
public UserResponse updateAndReturn(long id, UpdateRequest req) {
    User user = userRepo.findById(id).orElseThrow();
    user.applyUpdate(req);
    user.setVersion(user.getVersion() + 1);
    userRepo.save(user);
    return new UserResponse(user); // return WITH the new version to client
}

// Client sends version in next read. Check replica has that version.
public User getUserAtVersion(long id, long minVersion) {
    User user = replicaRepo.findById(id).orElseThrow();
    if (user.getVersion() < minVersion) {
        return primaryRepo.findById(id).orElseThrow(); // replica stale → escalate
    }
    return user;
}

// ── Solution 1: Local In-Process Cache (Caffeine) ─────────────────────
// Each application server caches celebrity data locally
// 100 app servers × cached locally = 100x DB read reduction

@Service
public class UserProfileService {

    private final Cache<Long, UserProfile> localCache = Caffeine.newBuilder()
        .maximumSize(10_000)
        .expireAfterWrite(Duration.ofSeconds(30))   // short TTL for freshness
        .recordStats()
        .build();

    public UserProfile getProfile(long userId) {
        return localCache.get(userId, id -> {
            // Cache miss → check Redis
            UserProfile cached = (UserProfile) redis.opsForValue().get("user:" + id);
            if (cached != null) return cached;
            // Redis miss → DB
            return userRepo.findById(id).orElseThrow();
        });
    }
}

// ── Solution 2: Cache Key Replication — copy hot key to N slots ───────
// Instead of one Redis key for celebrity, create N copies
// Distribute reads across all N copies

public UserProfile getCelebrityProfile(long userId) {
    int    replicas = 20;   // 20 Redis key copies
    int    slot     = (int) ((System.nanoTime() % replicas + replicas) % replicas);
    String key      = "user:" + userId + ":slot:" + slot;

    UserProfile profile = (UserProfile) redis.opsForValue().get(key);
    if (profile != null) return profile;

    // Cache miss → DB fetch + populate ALL slots
    profile = userRepo.findById(userId).orElseThrow();
    for (int i = 0; i < replicas; i++) {
        redis.opsForValue().set("user:" + userId + ":slot:" + i,
            profile, Duration.ofMinutes(5));
    }
    return profile;
}

// ── Solution 3: Detect hot keys + special treatment ───────────────────
// Count accesses, flag users crossing threshold as "celebrities"
// Apply different caching strategy for them

@Service
public class HotKeyDetector {

    private final ConcurrentHashMap<Long, LongAdder> accessCounts = new ConcurrentHashMap<>();
    private final Set<Long> hotKeys = ConcurrentHashMap.newKeySet();
    private static final long HOT_THRESHOLD = 10_000; // 10K/min = celebrity

    public UserProfile getProfile(long userId) {
        // Track access count
        accessCounts.computeIfAbsent(userId, k -> new LongAdder()).increment();

        if (hotKeys.contains(userId)) {
            return getFromLocalCache(userId);  // celebrity → local cache
        }

        long count = accessCounts.getOrDefault(userId, new LongAdder()).sum();
        if (count > HOT_THRESHOLD) {
            hotKeys.add(userId);  // promote to celebrity
            log.info("User {} promoted to hot key ({}  req/min)", userId, count);
        }

        return getFromRedis(userId);  // normal user → Redis
    }
}

CAP Theorem & Distributed Systems Trade-offs

C — Consistency

Every read receives the most recent write or an error. All nodes return the same data at the same time.

Requires coordination — slower writes

A — Availability

Every request receives a response (not an error). The system always responds — may return stale data.

Always responds — may not be fresh

P — Partition Tolerance

System continues operating even if network partition (nodes can't communicate). In distributed systems — P is always required.

Real networks always partition — P is non-negotiable

            Duration.ofHours(1));   // cached for fast lookup
    }
}

Consistent Hashing — How it Solves Resharding

// ── Problem with modulo hashing ───────────────────────────────────────
// shardId = hash(key) % N
// If N changes from 4 to 5: almost ALL keys move to different shards!
// With N=4→5: ~80% of keys need to be moved

// ── Consistent Hashing — minimal key movement ─────────────────────────
// Hash space is a ring (0 to 2^32)
// Nodes placed at points on the ring (by hash of node ID)
// Key → find nearest node clockwise
// Add node → only keys between new node and its predecessor move
// Remove node → only that node's keys move to next node
// Average key movement: k/n (k=keys, n=nodes) — optimal!

public class ConsistentHashRing<T> {

    private final TreeMap<Long, T> ring         = new TreeMap<>();
    private final int               virtualNodes;

    public ConsistentHashRing(int virtualNodes) {
        this.virtualNodes = virtualNodes;  // 150-200 vnodes per real node
    }

    public void addNode(T node) {
        for (int i = 0; i < virtualNodes; i++) {
            long hash = hash(node.toString() + "#" + i);
            ring.put(hash, node);
        }
    }

    public void removeNode(T node) {
        for (int i = 0; i < virtualNodes; i++) {
            long hash = hash(node.toString() + "#" + i);
            ring.remove(hash);
        }
    }

    public T getNode(String key) {
        if (ring.isEmpty()) return null;
        long hash = hash(key);
        // Find the first node at or after the key's hash
        Map.Entry<Long, T> entry = ring.ceilingEntry(hash);
        if (entry == null) entry = ring.firstEntry(); // wrap around ring
        return entry.getValue();
    }

    private long hash(String key) {
        // Use MurmurHash3 in production for better distribution
        return Hashing.murmur3_128().hashString(key, StandardCharsets.UTF_8).asLong();
    }
}

// Usage:
ConsistentHashRing<String> ring = new ConsistentHashRing<>(200);
ring.addNode("redis-1:6379");
ring.addNode("redis-2:6379");
ring.addNode("redis-3:6379");

String node = ring.getNode("user:12345"); // always same node for same key
ring.addNode("redis-4:6379");              // only ~25% of keys move
ring.removeNode("redis-1:6379");          // only redis-1's keys move to redis-2

// ── Idempotency Key — prevent duplicate processing ────────────────────
// Client sends unique idempotency key with each request
// Server checks if key already processed → return cached result
// Same as Stripe API idempotency keys

@PostMapping("/payments")
public PaymentResult createPayment(
        @RequestHeader("Idempotency-Key") String idempotencyKey,
        @RequestBody PaymentRequest request) {

    String cacheKey = "idem:" + idempotencyKey;
    PaymentResult cached = (PaymentResult) redis.opsForValue().get(cacheKey);
    if (cached != null) return cached;  // duplicate request → return same result

    PaymentResult result = paymentService.process(request);
    redis.opsForValue().set(cacheKey, result, Duration.ofHours(24));
    return result;
}

// ── Saga Pattern — distributed transactions ────────────────────────────
// Each step has a compensating transaction for rollback
// Choreography: each service reacts to events
// Orchestration: a saga orchestrator tells each service what to do

// Order Saga: PlaceOrder → ReserveInventory → ChargePayment → ConfirmOrder
// Failure at ChargePayment → ReleaseInventory → CancelOrder (compensate)

@Service
public class OrderSagaOrchestrator {
    public void execute(Order order) {
        try {
            inventoryService.reserve(order);
            try {
                paymentService.charge(order);
                orderService.confirm(order);
            } catch (PaymentException e) {
                inventoryService.release(order);  // compensate
                orderService.cancel(order);
            }
        } catch (InventoryException e) {
            orderService.cancel(order);
        }
    }
}

// ── Thundering Herd — cache stampede protection ───────────────────────
// When cache expires → all requests simultaneously hit DB

private final Map<String, Semaphore> keyLocks = new ConcurrentHashMap<>();

public Object getWithoutStampede(String key) {
    Object cached = cache.get(key);
    if (cached != null) return cached;

    Semaphore lock = keyLocks.computeIfAbsent(key, k -> new Semaphore(1));
    if (lock.tryAcquire()) {
        try {
            Object fresh = db.fetch(key);     // only ONE thread hits DB
            cache.set(key, fresh);
            return fresh;
        } finally { lock.release(); keyLocks.remove(key); }
    } else {
        lock.acquire();                        // others WAIT for winner to load
        lock.release();
        return cache.get(key);               // now in cache
    }
}
