2.1 CountryEnum.java

package com.india.architecture.context;
import lombok.Getter;
// Maps every country to: datasource key, ISO code, AWS region
@Getter
public enum CountryEnum {
    INDIA ("india", "IN", "ap-south-1"),
    US ("us", "US", "us-east-1"),
    UK ("uk", "UK", "eu-west-2"),
    GLOBAL("global", "GL", "us-east-1"); // default fallback
    private final String dsKey; // HikariCP datasource key
    private final String isoCode; // ISO 3166-1 alpha-2
    private final String awsRegion; // AWS deployment region
    CountryEnum(String dsKey, String isoCode, String awsRegion) {
        this.dsKey = dsKey;
        this.isoCode = isoCode;
        this.awsRegion = awsRegion;
    }
    // Accepts both "india" and "IN" — case insensitive
    public static CountryEnum fromHeader(String value) {
        if (value == null || value.isBlank()) return GLOBAL;
        for (CountryEnum c : values()) {
            if (c.dsKey.equalsIgnoreCase(value) ||
            c.isoCode.equalsIgnoreCase(value)) {
                return c;
            }
        }
        return GLOBAL;
    }
}

2.2 CountryContextHolder.java

package com.india.architecture.context;
// Thread-local country context — one slot per request thread
// Lifecycle: CountryFilter.set()   Service reads   Filter.clear()
public final class CountryContextHolder {
    private static final ThreadLocal<CountryEnum> CONTEXT =
    ThreadLocal.withInitial(() -> CountryEnum.GLOBAL);
    private CountryContextHolder() {} // prevent instantiation
    public static void set(CountryEnum country) {
        CONTEXT.set(country);
    }
    public static CountryEnum get() {
        return CONTEXT.get();
    }
    // Returns "india" | "us" | "uk" | "global"
    public static String dsKey() {
        return CONTEXT.get().getDsKey();
    }
    // Returns "IN" | "US" | "UK" | "GL"
    public static String isoCode() {
        return CONTEXT.get().getIsoCode();
    }
    // CRITICAL: call in finally block — prevents ThreadLocal memory leak
    public static void clear() {
        CONTEXT.remove();
    }
}

2.3 CountryRoutingDataSource.java

package com.india.architecture.routing;
import com.india.architecture.context.CountryContextHolder;
import lombok.extern.slf4j.Slf4j;
import org.springframework.jdbc.datasource.lookup.AbstractRoutingDataSource;
import javax.sql.DataSource;
// Spring calls determineCurrentLookupKey() before EVERY JDBC operation
@Slf4j
public class CountryRoutingDataSource extends AbstractRoutingDataSource {
    @Override
    protected Object determineCurrentLookupKey() {
        String key = CountryContextHolder.dsKey();
        log.debug("[DS-ROUTING] Routing JDBC   datasource key: '{}'", key);
        return key; // matches keys in DataSourceConfig targets map
    }
    @Override
    protected DataSource resolveSpecifiedDataSource(Object dataSource) {
        if (dataSource == null) {
            log.warn("[DS-ROUTING] Key not found, using GLOBAL fallback");
            return getResolvedDefaultDataSource();
        }
        return super.resolveSpecifiedDataSource(dataSource);
    }
}

    public DataSource indiaDataSource() {
        return DataSourceBuilder.create().build();
    }
    @Bean("usDS")
    @ConfigurationProperties(prefix = "datasources.us")
    public DataSource usDataSource() {
        return DataSourceBuilder.create().build();
    }
    @Bean("ukDS")
    @ConfigurationProperties(prefix = "datasources.uk")
    public DataSource ukDataSource() {
        return DataSourceBuilder.create().build();
    }
    @Bean("globalDS")
    @ConfigurationProperties(prefix = "datasources.global")
    public DataSource globalDataSource() {
        return DataSourceBuilder.create().build();
    }
    @Bean("routingDataSource")
    @Primary
    public DataSource routingDataSource(
    @Qualifier("indiaDS") DataSource india,
    @Qualifier("usDS") DataSource us,
    @Qualifier("ukDS") DataSource uk,
    @Qualifier("globalDS") DataSource global) {
        Map<Object, Object> targets = new HashMap<>();
        targets.put(CountryEnum.INDIA.getDsKey(), india); // "india"
        targets.put(CountryEnum.US.getDsKey(), us); // "us"
        targets.put(CountryEnum.UK.getDsKey(), uk); // "uk"
        targets.put(CountryEnum.GLOBAL.getDsKey(), global); // "global"
        CountryRoutingDataSource routing = new CountryRoutingDataSource();
        routing.setTargetDataSources(targets);
        routing.setDefaultTargetDataSource(global); // fallback
        routing.afterPropertiesSet();
        return routing;
    }
}

3.1 CountryFilter.java (Order = 1)

package com.india.architecture.filter;
import com.india.architecture.context.*;
import jakarta.servlet.*;
import jakarta.servlet.http.HttpServletRequest;
import lombok.extern.slf4j.Slf4j;
import org.slf4j.MDC;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;
import java.io.IOException;
import java.util.UUID;
// Runs FIRST on every HTTP request — sets country context
@Component
@Order(1)
@Slf4j
public class CountryFilter implements Filter {
    @Override
    public void doFilter(ServletRequest req, ServletResponse res,
    FilterChain chain) throws IOException, ServletException {
        HttpServletRequest request = (HttpServletRequest) req;
        // Priority: Header   Query Param   Default GLOBAL
        String raw = request.getHeader("X-Country-Code");
        if (raw == null || raw.isBlank()) {
            raw = request.getParameter("country");
        }
        CountryEnum country = CountryEnum.fromHeader(raw);
        // Correlation ID for distributed tracing
        String requestId = request.getHeader("X-Request-ID");
        if (requestId == null) requestId = UUID.randomUUID().toString();
        CountryContextHolder.set(country);
        MDC.put("country", country.getIsoCode());
        MDC.put("requestId", requestId);
        try {
            chain.doFilter(req, res);
        } finally {
            CountryContextHolder.clear(); // MUST — prevents ThreadLocal leak
            MDC.clear();
        }
    }
}

3.3 CountryDataService.java — Cache + Circuit Breaker

@Service
@RequiredArgsConstructor
@Slf4j
public class CountryDataService {
    private final CustomerRepository customerRepository;
    // READ: Cache-Aside (Redis L2) + Circuit Breaker
    @Cacheable(value = "customers",
    key = "T(com.india.architecture.context.CountryContextHolder)"
    + ".isoCode() + ':' + #state")
    @CircuitBreaker(name = "indiaDB", fallbackMethod = "fallbackByState")
    @Retry(name = "indiaDB")
    public List<Customer> getCustomersByState(String state) {
        String iso = CountryContextHolder.isoCode();
        return customerRepository.findByCountryCodeAndState(iso, state);
    }
    // WRITE: Evict cache + auto-set country from context
    @Transactional
    @CacheEvict(value = "customers", allEntries = true)
    public Customer save(Customer customer) {
        customer.setCountryCode(CountryContextHolder.isoCode());
        return customerRepository.save(customer);
    }
    // PAGINATED: always paginate — never load all rows!
    public Page<Customer> getPagedCustomers(Pageable pageable) {
        return customerRepository.findActiveByCountryCode(
        CountryContextHolder.isoCode(), pageable);
    }
    // CROSS-COUNTRY: switch context per loop iteration
    public Map<String, Long> countByAllCountries() {
        Map<String, Long> result = new LinkedHashMap<>();
        for (CountryEnum c : CountryEnum.values()) {
            if (c == CountryEnum.GLOBAL) continue;
            CountryContextHolder.set(c);
            try {
                result.put(c.name(),
                customerRepository.countByCountryCode(c.getIsoCode()));
            } finally {
                CountryContextHolder.clear(); // clear after each country
            }
        }
        return result;
    }
    // Fallback when circuit is OPEN
    public List<Customer> fallbackByState(String state, Exception ex) {
        log.error("[FALLBACK] India DB unavailable state={}: {}", state, ex.getMessage());
        return List.of(); // graceful empty response
    }
}

4. LLD — STARTUP LOADER & REST API

4.1 IndiaDataLoader.java (ApplicationRunner)

@Component
@RequiredArgsConstructor
@Slf4j
public class IndiaDataLoader implements ApplicationRunner {
    private final CustomerRepository customerRepository;
    private final CacheManager cacheManager;
    // Runs ONCE after Spring Boot startup
    // Pre-warms Redis cache with India reference data
    @Override
    public void run(ApplicationArguments args) {
        log.info("                                       ");
        log.info("  [STARTUP] Loading India data ");
        log.info("                                       ");
        CountryContextHolder.set(CountryEnum.INDIA);
        try {
            var customers = customerRepository.findByCountryCode("IN");
            var cache = cacheManager.getCache("customers");
            if (cache != null) {
                cache.put("IN:all", customers);
            }
            log.info("[STARTUP] Pre-warmed: {} India customers", customers.size());
        } catch (Exception ex) {
            // Non-fatal — app still starts even if DB is temporarily down
            log.warn("[STARTUP] Pre-warm failed (non-fatal): {}", ex.getMessage());
        } finally {
            CountryContextHolder.clear(); // Always clean up
        }
    }
}

5.3 Redis Cache-Aside Pattern (CacheConfig.java)

@Configuration
@EnableCaching
public class CacheConfig {
    @Bean @Primary
    public CacheManager redisCacheManager(RedisConnectionFactory factory) {
        RedisCacheConfiguration base = RedisCacheConfiguration.defaultCacheConfig()
        .entryTtl(Duration.ofMinutes(30))
        .disableCachingNullValues();
        Map<String, RedisCacheConfiguration> perCache = new HashMap<>();
        perCache.put("customers", base.entryTtl(Duration.ofMinutes(60)));
        perCache.put("products", base.entryTtl(Duration.ofMinutes(30)));
        perCache.put("masterData", base.entryTtl(Duration.ofHours(24)));
        return RedisCacheManager.builder(factory)
        .cacheDefaults(base)
        .withInitialCacheConfigurations(perCache)
        .build();
    }
    // L1: JVM Caffeine cache — fastest, per-instance, 5-min TTL
    @Bean("caffeineCacheManager")
    public CacheManager caffeineCacheManager() {
        CaffeineCacheManager mgr = new CaffeineCacheManager();
        mgr.setCaffeineSpec(CaffeineSpec.parse(
        "maximumSize=10000,expireAfterWrite=5m"));
        return mgr;
    }
}

// RateLimitFilter.java — Token bucket per (clientId + country)
private Bucket createBucket(String isoCode) {
    int limit = switch (isoCode) {
        case "IN" -> 1000; // India: 1000 req/min
        case "US" -> 2000; // US: 2000 req/min
        case "UK" -> 1500; // UK: 1500 req/min
        default -> 500;
    };
    return Bucket.builder()
    .addLimit(Bandwidth.classic(limit,
    Refill.greedy(limit, Duration.ofMinutes(1))))
    .build();
}

@Component
